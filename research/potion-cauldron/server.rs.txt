//! Unshipped experiment: registration is refused for vanilla water cauldrons.
use mod_api::{
    BlockUpdateFlags, Compound, Flow, Host, HostTransaction, Identifier, ItemStack, ServerHost,
    SoundSource, StatType, Tag,
};

const CONTENTS: &str = "minecraft:potion_contents";

pub fn register(registrar: &mut dyn mod_api::Registrar) {
    registrar
        .block_entity_type("water_cauldron")
        .blocks(&["minecraft:water_cauldron"])
        .build();
}

fn water() -> Compound {
    [("potion".into(), Tag::String("minecraft:water".into()))].into()
}
fn normalized(mut contents: Compound) -> Compound {
    contents.retain(|key, _| {
        matches!(
            key.to_string_lossy().as_ref(),
            "potion" | "custom_color" | "custom_effects" | "custom_name"
        )
    });
    if matches!(contents.get(&"custom_effects".into()), Some(Tag::List(list)) if list.is_empty()) {
        contents.remove(&"custom_effects".into());
    }
    contents
}
fn potion(host: &dyn Host, stack: &ItemStack) -> Option<Compound> {
    let value = stack
        .components()
        .and_then(|patch| patch.as_nbt().get(&CONTENTS.into()))
        .cloned()
        .or_else(|| host.item_default_component(stack.item(), CONTENTS))?;
    let Tag::Compound(contents) = value else {
        return None;
    };
    let contents = normalized(contents);
    (contents.contains_key(&"potion".into()) || matches!(contents.get(&"custom_effects".into()), Some(Tag::List(list)) if !list.is_empty())).then_some(contents)
}
fn with_count(stack: &ItemStack, count: u32) -> Option<ItemStack> {
    let mut result = ItemStack::new(stack.item(), count)?;
    if let Some(patch) = stack.components() {
        result.set_components(patch.into()).ok()?;
    }
    Some(result)
}
fn make_stack(
    host: &dyn Host,
    id: &str,
    count: u32,
    contents: Option<&Compound>,
) -> Option<ItemStack> {
    let key = host.item_key(&Identifier::parse(id).ok()?)?;
    let mut stack = ItemStack::new(key, count)?;
    if let Some(contents) = contents {
        stack
            .apply_components(&[(CONTENTS.into(), Tag::Compound(contents.clone()))].into())
            .ok()?;
    }
    Some(stack)
}

#[derive(Clone, Debug)]
enum Action {
    Pour {
        contents: Compound,
        bottle: String,
        fill: u8,
        mixed: bool,
    },
    Fill {
        contents: Compound,
        bottle: String,
        fill: u8,
    },
    Dip {
        contents: Compound,
        count: u32,
        fill: u8,
    },
}
fn plan(
    item: &str,
    count: u32,
    fill: u8,
    saved: &Compound,
    incoming: Option<Compound>,
) -> Option<Action> {
    let stored = match saved.get(&"potionContents".into()) {
        Some(Tag::Compound(value)) => normalized(value.clone()),
        _ => water(),
    };
    let bottle = match saved.get(&"lastPotionItemUsed".into()) {
        Some(Tag::String(value)) => value.to_string_lossy(),
        _ => "minecraft:potion".into(),
    };
    match item {
        "minecraft:potion" | "minecraft:splash_potion" | "minecraft:lingering_potion" => {
            let contents = incoming?;
            if fill == 3 && stored == contents && !saved.contains_key(&"dyeColor".into()) {
                return None;
            }
            let mixed = fill > 0 && stored != contents;
            Some(Action::Pour {
                contents,
                bottle: item.into(),
                fill: if mixed { 0 } else { (fill + 1).min(3) },
                mixed,
            })
        }
        "minecraft:glass_bottle" if fill > 0 && stored != water() && !saved.contains_key(&"dyeColor".into()) => Some(Action::Fill {
            contents: stored,
            bottle,
            fill: fill - 1,
        }),
        "minecraft:arrow" if fill > 0 && stored != water() => {
            let count = count.min(if fill < 3 { u32::from(fill) * 16 } else { 64 });
            let used = count.div_ceil(16).min(3) as u8;
            // Preserve the checked-out LegacyBlockBehaviors formula, including
            // its remaining-level choice: nonempty uses `used`, not `fill-used`.
            Some(Action::Dip {
                contents: stored,
                count,
                fill: if fill == used { 0 } else { used },
            })
        }
        _ => None,
    }
}

pub fn click(host: &mut dyn ServerHost, click: &mod_api::BlockClick<'_>) -> Flow {
    let Some(player) = host.player(click.player) else {
        return Flow::Continue;
    };
    let Some(level) = player.level(host) else {
        return Flow::Continue;
    };
    let Some(block) = level.block(host, click.pos) else {
        return Flow::Continue;
    };
    let id = block.id(host);
    if id != "minecraft:cauldron" && id != "minecraft:water_cauldron" {
        return Flow::Continue;
    }
    let Some(stack) = click.stack else {
        return Flow::Continue;
    };
    let item = stack.item().id(host);
    let fill = level
        .block_property(host, click.pos, "level")
        .and_then(|value| value.parse::<u8>().ok())
        .unwrap_or(0);
    let saved = host
        .block_entity(level, click.pos)
        .and_then(|entity| entity.data(host))
        .unwrap_or_default();
    let Some(action) = plan(&item, stack.count(), fill, &saved, potion(host, stack)) else {
        return Flow::Continue;
    };
    let Some(living) = host.living(player.entity()) else {
        return Flow::Continue;
    };
    if living.hand_item(host, click.hand).as_ref() != Some(stack) {
        return Flow::Continue;
    }
    let infinite = player.has_infinite_materials(host).unwrap_or(false);
    let result: Result<(), String> = host.transaction(|host, _| {
        let (new_fill, contents, last_bottle, consumed, output, sound) = match &action {
            Action::Pour {
                contents,
                bottle,
                fill,
                mixed,
            } => (
                *fill,
                contents,
                bottle.as_str(),
                1,
                make_stack(host, "minecraft:glass_bottle", 1, None),
                if *mixed {
                    "minecraft:block.fire.extinguish"
                } else {
                    "minecraft:item.bottle.empty"
                },
            ),
            Action::Fill {
                contents,
                bottle,
                fill,
            } => (
                *fill,
                contents,
                bottle.as_str(),
                1,
                make_stack(host, bottle, 1, Some(contents)),
                "minecraft:item.bottle.fill",
            ),
            Action::Dip {
                contents,
                count,
                fill,
            } => (
                *fill,
                contents,
                "minecraft:potion",
                *count,
                make_stack(host, "minecraft:tipped_arrow", *count, Some(contents)),
                "",
            ),
        };
        let output = output.ok_or("output item/component unavailable")?;
        let state_name = if new_fill == 0 {
            "minecraft:cauldron".into()
        } else {
            format!("minecraft:water_cauldron[level={new_fill}]")
        };
        let state = host
            .block_state_id(&state_name)
            .ok_or("cauldron state unavailable")?;
        level
            .set_block(
                host,
                click.pos,
                state,
                BlockUpdateFlags::NEIGHBORS | BlockUpdateFlags::CLIENTS,
            )
            .map_err(|e| format!("{e:?}"))?;
        if new_fill > 0 {
            let entity = host
                .block_entity(level, click.pos)
                .ok_or("water cauldron has no saved block entity")?;
            let data = [
                ("potionContents".into(), Tag::Compound(contents.clone())),
                ("lastPotionItemUsed".into(), Tag::String(last_bottle.into())),
            ]
            .into();
            entity
                .replace_data(host, data)
                .map_err(|e| format!("{e:?}"))?;
        }
        if !infinite && matches!(action, Action::Dip { .. }) {
            let remaining = if stack.count() > consumed { with_count(stack, stack.count() - consumed) } else { None };
            living.set_hand_item(host, click.hand, remaining).map_err(|e| format!("{e:?}"))?;
            player.give_stack(host, output).map_err(|e| format!("{e:?}"))?;
        } else if !infinite {
            if stack.count() <= consumed {
                living
                    .set_hand_item(host, click.hand, Some(output))
                    .map_err(|e| format!("{e:?}"))?;
            } else {
                living
                    .set_hand_item(
                        host,
                        click.hand,
                        with_count(stack, stack.count() - consumed),
                    )
                    .map_err(|e| format!("{e:?}"))?;
                player
                    .give_stack(host, output)
                    .map_err(|e| format!("{e:?}"))?;
            }
        } else {
            let already_has_output = !matches!(action, Action::Dip { .. })
                && player
                    .inventory(host)
                    .is_some_and(|slots| slots.iter().any(|slot| slot.is_stackable_with(&output)));
            if !already_has_output {
                player
                    .give_stack(host, output)
                    .map_err(|e| format!("{e:?}"))?;
            }
        }
        player.award_stat(host, StatType::Custom, "minecraft:use_cauldron", 1);
        player.award_stat(host, StatType::Used, &item, 1);
        if !sound.is_empty() {
            level.play_sound(
                host,
                sound,
                click.pos.map(|v| f64::from(v) + 0.5),
                SoundSource::Blocks,
                1.0,
                1.0,
            );
        }
        level.game_event(
            host,
            if matches!(action, Action::Fill { .. }) {
                "minecraft:fluid_pickup"
            } else if matches!(action, Action::Dip { .. }) {
                "minecraft:block_change"
            } else {
                "minecraft:fluid_place"
            },
            click.pos.map(|v| f64::from(v) + 0.5),
            mod_api::GameEventContext {
                source_entity: Some(player.entity()),
                affected_state: Some(state),
            },
        );
        Ok(())
    });
    match result {
        Ok(()) => Flow::Cancel,
        Err(error) => {
            eprintln!("Legacy Horizon: potion cauldron refused: {error}");
            Flow::Continue
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    fn poison() -> Compound {
        [("potion".into(), Tag::String("minecraft:poison".into()))].into()
    }
    #[test]
    fn potion_fill_mix_and_dip_follow_source() {
        let data: Compound = [("potionContents".into(), Tag::Compound(poison()))].into();
        assert!(matches!(
            plan("minecraft:potion", 1, 0, &Compound::new(), Some(poison())),
            Some(Action::Pour {
                fill: 1,
                mixed: false,
                ..
            })
        ));
        assert!(matches!(
            plan("minecraft:potion", 1, 3, &data, Some(water())),
            Some(Action::Pour {
                fill: 0,
                mixed: true,
                ..
            })
        ));
        assert!(plan("minecraft:potion", 1, 3, &data, Some(poison())).is_none());
        assert!(matches!(
            plan("minecraft:arrow", 64, 3, &data, None),
            Some(Action::Dip {
                count: 64,
                fill: 0,
                ..
            })
        ));
        assert!(matches!(
            plan("minecraft:arrow", 1, 3, &data, None),
            Some(Action::Dip {
                count: 1,
                fill: 1,
                ..
            })
        ));
    }
}
